-- 1. Create Email Invite (Owner/Admin only)
POST /api/workspaces/{workspaceId}/invitations
Request:
{
  "email": "colleague@company.com",
  "role": "MEMBER" // or "ADMIN"
}
Response 201:
{
  "invitationId": 101,
  "email": "colleague@company.com",
  "token": "inv_a8f93bc1d9e2...",
  "role": "MEMBER",
  "expiresAt": "2026-08-25T15:00:00Z"
}

-- 2. Create Shareable Join Link (Owner/Admin only)
POST /api/workspaces/{workspaceId}/join-link
Request:
{
  "role": "MEMBER",
  "maxUses": 50,
  "expiresInDays": 30
}
Response 201:
{
  "joinUrl": "https://app.domain.com/join/join_7c2e1f...",
  "token": "join_7c2e1f...",
  "maxUses": 50,
  "expiresAt": "2026-09-17T15:00:00Z"
}

-- 3. Public Preview Invitation
GET /api/invitations/{token}
Response 200:
{
  "workspaceName": "Acme Corp",
  "workspaceSlug": "acme",
  "inviterName": "Alice",
  "targetEmail": "colleague@company.com", // masked if public
  "role": "MEMBER",
  "isValid": true
}

-- 4. Accept Invitation (Authenticated User)
POST /api/invitations/{token}/accept
Response 200:
{
  "workspaceId": 1,
  "workspaceSlug": "acme",
  "memberId": 5,
  "role": "MEMBER"
}
